---

- 将`context`大写
	- 问题分析：
		- 怎么确定循环次数
			- 一个一个数，人为确定
			- 使用`\0`结尾，标记一个字符串，读到停止
```asm
assume cs:code,ds:data
data segment
	db 'context',0
data ends
code segment
main:
	mov ax,data
	mov ds,ax
	
	mov si,0
	call fun
	mov ax,4c00h
	int 21h
	
fun:
	mov ch,0
	mov cl,[si]
	jcxz ok
	and byte ptr [si],11011111b ;小写——》大写
	inc si
	jmp short fun
ok:
	ret
code ends
end main
```
