# 一.万能查询

数据库查询出的<mark style="background: #FF5582A6;">每一条数据</mark>基本上都会<mark style="background: #FF5582A6;">封装为一个对象</mark>，数据库中的<mark style="background: #FF5582A6;">每一个字段值</mark>都会存储在对象<mark style="background: #FF5582A6;">相应的属性</mark>中。如果<mark style="background: #FF5582A6;">查询结果的每一个字段都与对象中的属性名保持一致</mark>，那么就可以使用<mark style="background: #FF5582A6;">反射来完成万能查询</mark>。

注：
1. 查询结果resultset类的列名称由别名定

# 二.设计万能查询

## 2.1普通查询

```java
package com.jsp.jdbc;  
  
import java.util.*;  
import java.sql.*;  
  
import com.mysql.jdbc.*;  
  
public class mysql_reflection {  
    public static void main(String[] args){  
        String url = "jdbc:mysql://localhost:3306/student?serverTimezone=Asia/Shanghai";//8.0以后要求添加时区  
        String username = "root";  
        String password = "root";  
        List<agent> list = new ArrayList<agent>();  
        try {  
            Class.forName("com.mysql.cj.jdbc.Driver");  
            Connection con = DriverManager.getConnection(url,username,password);  
            String sql = "select id,name,region_id from agent where name like ?";  
           PreparedStatement ps = con.prepareStatement(sql);  
           ps.setString(1,"%小米%");  
           ResultSet re = ps.executeQuery();  
           while(re.next()){  
               agent a = new agent();  
               a.setId(re.getInt("id"));  
               a.setName(re.getString("name"));  
               a.setRegion(re.getInt("region_id"));  
               list.add(a);  
           }  
           list.forEach(System.out::println);  
           re.close();  
           ps.close();  
           con.close();  
        } catch (Exception e) {  
            throw new RuntimeException(e);  
        }  
    }  
}
```

## 2.3万能查询

相比两个不同的查询，哪里不同提取出来，换成参数：
```java
package com.jsp.jdbc;  
  
import java.lang.reflect.*;  
import java.util.*;  
import java.sql.*;  
  
import com.mysql.jdbc.*;  
  
public class mysql_reflection {  
    public static void main(String[] args){  
        String sql ="select id,name,number,price,agent_id from goods where name LIKE ? AND price > ?";  
        Object[] params ={  
                "%小米%",  
                1000  
        };  
        Query(sql,goods.class,params);  
    }  
  
    public static<T>  List<T> Query(String sql ,Class<T> clazz,Object ...params){//不定长的参数要放在最后  
        String url = "jdbc:mysql://localhost:3306/student?serverTimezone=Asia/Shanghai";//8.0以后要求添加时区  
        String username = "root";  
        String password = "root";  
        List<T> list = new ArrayList<T>();  
        try {  
            Class.forName("com.mysql.cj.jdbc.Driver");  
            Connection con = DriverManager.getConnection(url,username,password);  
            //sql语句不确定提取，做为参数传入  
            PreparedStatement ps = con.prepareStatement(sql);  
            if(params != null && params.length > 0){//放入预处理的参数不确定提取  
                for(int i = 0 ; i < params.length ;i++){  
                    ps.setObject(i+1,params[i]);  
                }  
            }  
            ResultSet re = ps.executeQuery();  
            while(re.next()){//对象不知道用什么存储——》反射  
                T o = clazz.getConstructor().newInstance();//获得构造器和实例  
                Field[] fields = clazz.getDeclaredFields();  
                for(Field field :fields){  
                    String fieldName = field.getName();//注意若字段名与数据库的列名一致可以取列值  
                    String methodName = "set" + fieldName.substring(0,1).toUpperCase() + fieldName.substring(1);  
                    Method method = clazz.getDeclaredMethod(methodName,field.getType());  
                    try {//可能找不到对应的数据，列名与属性名不一致  
                        Object value = re.getObject(field.getName(),field.getType());//数据库内的列值找到以后根据提供的class转化  
                        method.invoke(o,value);  
                    } catch (Exception e) {  
                          
                    }  
                }  
                list.add(o);  
            }  
            list.forEach(System.out::println);  
            re.close();  
            ps.close();  
            con.close();  
        } catch (Exception e) {  
            throw new RuntimeException(e);  
        }  
        return list;  
    }  
}
```

# 三.万能更新

封装最终方案：

```java
package com.jsp.jdbc;  
  
import java.lang.reflect.*;  
import java.util.*;  
import java.sql.*;  
  
import com.mysql.jdbc.*;  
  
public class mysql_reflection {  
    private static final String url = "jdbc:mysql://localhost:3306/student?serverTimezone=Asia/Shanghai";//8.0以后要求添加时区  
    private static final String username = "root";  
    private static final String password = "root";  
    static {  
        try {  
            Class.forName("com.mysql.cj.jdbc.Driver");  
        } catch (ClassNotFoundException e) {  
            System.out.println("驱动加载失败");  
        }  
    }  
    public static void main(String[] args){  
        String sql ="select id,name,number,price,agent_id agentId from goods where name LIKE ? AND price > ?";  
        Object[] params ={  
                "%小米%",  
                1000  
        };  
        Query(sql,goods.class,params);  
    }  
  
    public static<T>  List<T> Query(String sql ,Class<T> clazz,Object ...params){//不定长的参数要放在最后  
        List<T> list = new ArrayList<T>();  
        Connection con = null;//放在外面避免局部限制  
        PreparedStatement ps = null;  
        ResultSet re = null;  
        try {  
            con = DriverManager.getConnection(url,username,password);  
            //sql语句不确定提取，做为参数传入  
            ps = getPreparedStatement(con,sql,params);  
            re = ps.executeQuery();  
            while(re.next()){//对象不知道用什么存储——》反射  
                T o = createInstance(clazz,re);  
                list.add(o);  
            }  
            list.forEach(System.out::println);  
        } catch (Exception e) {  
            throw new RuntimeException(e);  
        }finally{  
            Close(re,ps,con);  
        }  
        return list;  
    }  
    public static<T> T createInstance(Class<T> clazz ,ResultSet re) throws Exception{  
        T o = clazz.getConstructor().newInstance();//获得构造器和实例  
        Field[] fields = clazz.getDeclaredFields();  
        for(Field field :fields){  
            String fieldName = field.getName();//注意若字段名与数据库的列名一致可以取列值  
            String methodName = "set" + fieldName.substring(0,1).toUpperCase() + fieldName.substring(1);  
            Method method = clazz.getDeclaredMethod(methodName,field.getType());  
            try {//可能找不到对应的数据，列名与属性名不一致  
                Object value = re.getObject(field.getName(),field.getType());//数据库内的列值找到以后根据提供的class转化  
                method.invoke(o,value);  
            } catch (Exception e) {  
                e.printStackTrace();  
            }  
        }  
        return o;  
    }  
    public static int update(String sql , Object... params){  
        int count = 0;  
        Connection con = null;  
        PreparedStatement ps =null;  
        try{  
            con = DriverManager.getConnection(url,username,password);  
            //sql语句不确定提取，做为参数传入  
            ps = getPreparedStatement(con,sql,params);  
            count = ps.executeUpdate();  
            ps.close();  
            con.close();  
        }catch(Exception e){  
            e.printStackTrace();  
        }finally{  
            Close(ps,con);  
        }  
        return count;  
    }  
    public static PreparedStatement getPreparedStatement(Connection con , String sql,Object... params)throws Exception{  
            PreparedStatement ps = con.prepareStatement(sql);  
            if(params != null && params.length > 0){//放入预处理的参数不确定提取  
                for(int i = 0 ; i < params.length ;i++){  
                    ps.setObject(i+1,params[i]);  
                }  
            }  
            return ps;  
    }  
    public static void Close(AutoCloseable ...closeables){  
        if(closeables != null && closeables.length >0){  
            for(AutoCloseable closeable : closeables){  
                if(closeable != null){  
                    try {  
                        closeable.close();  
                    } catch (Exception e) {  
                        throw new RuntimeException(e);  
                    }  
                }  
            }  
        }  
    }  
}
```
