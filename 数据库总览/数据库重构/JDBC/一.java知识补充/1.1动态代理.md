import java.lang.reflect.Method;  
import java.lang.reflect.Proxy;  
  
  
public class ProxyUtil {  
    /*  
    * 使用代理对象：  
    *   1.获取代理对象  
    *   代理对象：public  start  createProxy() {  
    *   2.调用代理对象的唱歌方法  
    *   代理对象.唱歌方法 ——》 调用invoke方法  
    * */    public  static start  createProxy(bigstart bigstart) {  
  
        /*  
         * 方法的作用：  
         *   给一个明星对象，创建一个代理  
         *         * 形参：  
         *     给代理的对象  
         * 返回值：  
         *   给明星创建的代理，使用的是接口确定的方法，代理以后就不是bigstart这个对象了  
         * */  
  
        /*         * 第一个参数：用于指定哪个类加载器，去加载生成的代理类  
         * 第二个参数：指定接口，知道有什么方法代理,是一个数组，可以实现多个接口  
         * 第三个参数：指定生成的代理对象要干什么事,使用一个接口来重写实现类的对象  
         * */        start s = (start) Proxy.newProxyInstance(ProxyUtil.class.getClassLoader(), new Class[]{start.class}, new InvocationHandler() {  
  
            // 指明代理做什么事  
            @Override  
            /*  
            * 第一个参数：代理的对象  
            * 第二个参数：要运行的方法 ，即代理对象调用的方法  
            * 第三个参数：运行代理方法时传递的参数  
            * */            public Object invoke(Object proxy, Method method, Object[] args) throws Throwable {  
                //确定方法是什么，进行修改代码  
                if("sing".equals(method.getName())) {  
                    System.out.println("准备话筒,收钱");  
  
                }else if("dance".equals(method.getName())) {  
                    System.out.println("准备场地，收钱");  
                }  
                //调用方法  
                return method.invoke(bigstart,args);  
            }  
        });  
        return s;  
    }  
}

## 2.2获取class

---

- 获取`class`:
	- `class.forName("全类名")`
		- `class一个类，是类的类`,相当于获取这个类的实例
		- 全类名：报名 + 类名
		- 获得阶段：
			- 源码阶段：
				- 在磁盘中执行`java文件 ——》class文件`
	- `类名.class`:
		- 获取阶段：
			- 加载阶段：
				- 将`.class文件加载到内存中`
	- `对象.getClass()`
		- 获得阶段:
			- 运行阶段：
				- `创建了一个类的对象`

```java
import static java.lang.Class.forName;

public class relectDemo {
    public static void main(String[] args) throws ClassNotFoundException {
        /*
        * 三种获得class的方式：
        *   1.class.forName("类全名")
        *   2.类名.class
        *   3.对象.getclass();
        * */

        //1.方式1:常用
        Class clazz = Class.forName("bigstart");
        System.out.println(clazz);
        
        //2.方式2：常作为参数传递
        Class clazz2 = bigstart.class;
        //3.方式3：当有类对象时使用
        bigstart s = new bigstart("1432");
        Class clazz3 = s.getClass();
        
        //true
        System.out.println(clazz3 == clazz);
        System.out.println(clazz == clazz2);
    }
}

```

## 2.3获取构造方法

---

- 万物皆对象
	- `a.class`是`class`类的对象
	- 构造方法:`Constructor`
	- 字段:`Field`
	- 成员方法:`Method`

- `Constructor`:
	- 构造器获得时声明中无需自己写泛形类型
	- `class中的方法`：
		- `Constructor<?>[] getConstructors()`:
			- 返回所有公共构造方法对象的数组
		- `Constructor<?>[] getDelaredConstructors():`
			- 获得所有的构造方法对象的数组
		- `Constructor<T>[] getConstructor(Class<?>...parameterTypes)`:
			- 参数：
				- 对应参数的`class`，指定要求返回的构造方法的参数和参数一样
			- 获得单个公共构造方法,若是获得非公共的回报错
		- `Constructor<T>getDelaredConstructor(Class<?>...parameterTypes)`:
			- 参数：
				- 对应参数的`class`，指定要求返回的构造方法的参数和参数一样
			- 获得单个构造方法

```java
import java.lang.reflect.Constructor;

import static java.lang.Class.forName;

public class relectDemo {
    public static void main(String[] args) throws Throwable {
        /*
        * 三种获得class的方式：
        *   1.class.forName("类全名")
        *   2.类名.class
        *   3.对象.getclass();
        * */

        //1.方式1
        Class clazz = Class.forName("bigstart");
        System.out.println(clazz);

        //2.方式2
        Class clazz2 = bigstart.class;
        //3.方式3
        bigstart s = new bigstart("1432");
        Class clazz3 = s.getClass();

        //true
        System.out.println(clazz3 == clazz);
        System.out.println(clazz == clazz2);

        Constructor con = clazz.getConstructor(String.class);
        System.out.println(con);
        Constructor[] con2 = clazz2.getDeclaredConstructors();
        for(Constructor c : con2) {
            System.out.println(c);
        }
        Constructor con3 = clazz.getDeclaredConstructor(String.class,int.class);
        System.out.println(con3);
        Constructor[] con4 = clazz3.getConstructors();
        for(Constructor c : con4) {
            System.out.println(c);
        }
    }
}

```

- 剖析构造方法：
	- `getModifiers`:
		- 返回值：整数
			- `private`:2
			- `public`:1
			- `abstract`:1024
			- `provcted`:4
		- 可能编程没有什么用，但是对于`idea`来说有主要意义，私有方法不允许调用，显示提示
	- `Parameter[] getParameters`:
		- 获得所有参数
	- `newInstance(args[])`:
		- 参数：
			- 和对应的构造方法的参数保持一致
		- 构造一个对象，注意权限问题，`private`无法使用，能获得构造器，但是不能用
	- `setAcessible(true)`:
		- 忽略这个构造器的权限
		- 返回值：
			- 返回`class<T>的T类型`

```java

```

## 2.4成员变量

---

- `class关于field的方法`
	- 获得`Field对象`:
		- `Field[] getFields()`:
			- 返回公共成员变量的数组
		- `Field[] getDeclaredFields()`:
			- 返回所有成员变量
		- `Field[]getField(String name)`:
			- 返回`name`的公共成员变量
		- `Field getDeclaredField(String name)`:
			- 返回`name`所有的的成员变量

- 使用`Field对象`的方法：
	- `void set(Object obj,Object value):`
		- 赋值
		- 参数：修改哪个对象的什么值
	- `Object get(Object obj)`:
		- 获取值
		- 参数：实例对象，指定获得哪个对象的值
	- `String getName()`:
		- 获取字段名字
	- `Class<?> getType()`
		- 获取字段的类型
	- `int getModifilers`
		- 获取权限修饰符

```java
import java.lang.reflect.Field;

public class test_field {
    public static void main(String[] args) throws Throwable{
        //获得class对象
        Class clazz = bigstart.class;
        //1.获取单个字段变量
        //若是想用getField获得私有的，可以设置暂停权限检查
        Field f1 = clazz.getDeclaredField("name");
        //2.获得单个字段变量的名字
        String name = f1.getName();
        //3.获得修饰符
        int modifilers = f1.getModifiers();
        //4.获得字段的类型
        Class<?> c = f1.getType();
        System.out.println(name);
        System.out.println(f1);
        System.out.println(modifilers);
        System.out.println(c);
        //5.使用字段修改指定对象的值和获得指定对象的值
        bigstart s = new bigstart("t2");
        f1.get(s);
        System.out.println(s.name);
        f1.set(s,"tom");
        System.out.println(s.name);

    }
}

```

## 2.5成员方法

---

- `class的关于成员方法的方法`:
	- `Method[] getMethods()`:
		- 返回公共成员的对象
		- 可以获得父类的所有方法
	- `Method[] getDeclaredMethods()`
		- 返回所有成员方法的对象
		- 不可以获得父类的方法
	- `Method getMethod(String name,Class<?>....parameterTypes)`
	- `Method getDeclaredMethod(String name,Class<?>....parameterTypes)`
- `Method类中用与使用方法的方法`:
	- `Object invoke(Object obj,Object .... args)`
	- `setAccess(true)`:
		- 忽略权限检查，直接使用
	- `getName()`:
		- 获得方法的名字
	- `getModifilers()`:
		- 获得修饰符
	- `Parameters[] getParamerters`
		- 获得形参
	- `Class[] exceptionTypes`:
		- 获得方法可能抛出的异常
	- `getReturnType()`：获取原始返回类型
