```javascript
//补充
pop();            //删除数组末尾最后一个元素，返回数组长度

```

注：
1. 向匿名创建，由浏览器调用的函数使用——》回调函数，回调函数的传入是由函数严格限制的
2. 浏览器会将遍历到的函数以实参形式传入回调函数中，第一个参数是数组当前正在遍历的元素——》`value`，第二个参数是索引——》`index`，第三个参数是数组/对象，即正在遍历的数组——》`object`

```java
let num1= [1, 2, 3]

let length = num1.push(4, 5);//一次放入多个元素至数组中

console.log("数组长度：" + length);

let num2 = [6, 7, 8];

let num3 = num1.concat(num2);//将数组num2与num1进行在新数组中进行拼接，num2在num1之后

console.log("拼接后：" +num3);

num3.splice(2, 1); //将数组num3从下标为2的位置删除1个元素

console.log("删除元素后：" + num3);

num3.splice(3, 2, 10, 20, 30); //将数组num3从下标为3的位置删除2个元素，然后将10,20,30从

删除位置添加到数组中

console.log("删除元素的同时增加元素：" + num3)

let str = num3.join(","); //将数组num3中所有元素使用","拼接起来，1，2，3不会出现在最后一个元素后面

console.log(str);

num3.foreach(function(a,b,c){
	console.log("a" + a);
	console.log("b" + b);
	console.log("c" + c);
});
```

## 4.3数组练习

- 去重：
```javascript
for(let i = 0 ; i < arr.length ; i++){
	for(let j= i + 1; j < arr.length ; j++){
		if(arr[i] == arr[j]){
			arr.splice(j,1);//可能因为删除后，原来相邻的元素顶上来，顶替删除位置，跳过了判断
			j--;//判断删除位置的顶替元素
		}
	}
}
```

# 九.元事件
