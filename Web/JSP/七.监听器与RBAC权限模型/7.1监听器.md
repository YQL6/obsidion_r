# 一.监听器

## 1.1什么是监听器

监听器顾名思义就是监听某种事件的发生，一旦<mark style="background: #FF5582A6;">监听的事件触发</mark>，那么监听器就将<mark style="background: #FF5582A6;">开始执行</mark>。例如：在 上课的时候，老师会观察每一位学生的听课情况，如果有学生上课打瞌睡，那么老师就会提醒他。这个 场景中，老师就是一个监听器，监听的是学生是否打瞌睡，一旦学生出现打瞌睡的情况，监听器就开始 执行（老师提醒学生）

## 1.2上下文监听器

监听上下文的监听器

作用：获得配置里面对上下文的配置，配置可以对应一个文件，读取里面内容，营造运行环境，<mark style="background: #FF5582A6;">做为数据源</mark>

示例：
ApplicationContextListener:
```java
package com.jsp.listener;  
  
import javax.servlet.ServletContext;  
import javax.servlet.ServletContextEvent;  
import javax.servlet.ServletContextListener;  
import javax.servlet.annotation.WebListener;  
import java.io.*;  
import java.util.Properties;  
  
@WebListener  
public class ApplicationListener  implements ServletContextListener {  
    @Override  
    public void contextDestroyed(ServletContextEvent sce) {  
        System.out.println("监听器关闭释放资源");  
    }  
  
    @Override  
    public void contextInitialized(ServletContextEvent sce) {  
        System.out.println("监听器开始初始化上下文");  
        ServletContext servletcontext = sce.getServletContext();  
        String jdbcConfig = servletcontext.getInitParameter("jdbcConfig");  
        InputStream is = this.getClass().getClassLoader().getResourceAsStream(jdbcConfig);//依据列类路径找到文件位置  
        Properties prop = new Properties();  
        try {  
            prop.load(is);//依靠is流获得文件数据，加载到propeperties的类中  
            System.out.println(prop);  
        } catch (IOException e) {  
            throw new RuntimeException(e);  
        }  
    }  
}
```

4. this.getClass().getClassLoader().getResourceAsStream(jdbcConfig)，这个就是获得本claas类然后获得其类加载器，然后类加载器获得jdbcConfig的文件存储位置的输入流
5. 输入流是对本程序来说，不是对jdbcCofig文件来说

# 二.properties初步配置规则

规则：
1.  **点号之前**（如 `jdbc`）：代表**“领域/模块/类别”**（像一个文件夹）。
2. **点号之后**（如 `url`）：代表**“具体属性”**（像一个文件）。
3. 默认用`.`区分不同领域的内容，本质上依旧是命名

package com.jsp.jdbc.handler;  
  
import org.apache.commons.beanutils.BeanUtils;  
  
import java.sql.ResultSet;  
import java.sql.ResultSetMetaData;  
import java.sql.SQLException;  
import java.util.*;  
  
public class MultiResultHandler<T> implements ResultHandler<List<T>>{  
    private Class<T> clazz = null;  
  
    public MultiResultHandler(Class<T> clazz){  
        this.clazz = clazz;  
    }  
  
    @Override  
    public List<T> handle(ResultSet rs) throws SQLException {  
        List<T> list = new ArrayList<>();  
        while(rs.next()){  
            try {  
                Map<String, Object> map = new HashMap<>();  
                T t = clazz.getConstructor().newInstance();  
                ResultSetMetaData remd = rs.getMetaData();  
                int columnCount = remd.getColumnCount();  
                for(int i = 1; i <= columnCount ; i++){  
                    String columnLabel = remd.getColumnLabel(i);  
                    Object columnValue = rs.getObject(columnLabel);  
                    map.put(columnLabel, columnValue);  
                }  
                BeanUtils.populate(t,map);  
                list.add(t);  
            } catch (Exception e) {  
                throw new RuntimeException(e);  
            }  
        }  
        return list;  
    }  
}

package com.jsp.jdbc.handler;  
  
import org.apache.commons.beanutils.BeanUtils;  
  
import java.sql.ResultSet;  
import java.sql.ResultSetMetaData;  
import java.sql.SQLException;  
import java.util.HashMap;  
import java.util.Map;  
  
public class SingleHandler<T> implements ResultHandler<T>{//单个数据的处理  
    private Class<T> clazz = null;  
  
    public SingleHandler(Class<T> clazz){  
        this.clazz = clazz;  
    }  
  
    @Override  
    public T handle(ResultSet rs) throws SQLException {  
        int count = 0;  
        while(rs.next()){  
            count++;  
            if(count > 1){  
                throw new RuntimeException("查询结果含有多条数据" + count);  
            }  
            try{  
                T t = clazz.getConstructor().newInstance();  
                Map<String, Object> map = new HashMap<String , Object>();  
                ResultSetMetaData remd = rs.getMetaData();  
                int columnCount = remd.getColumnCount();  
                for(int i = 1; i <= columnCount ; i++){  
                    String columnLabel = remd.getColumnLabel(i);  
                    Object columnValue = rs.getObject(columnLabel);  
                    map.put(columnLabel, columnValue);  
                }  
                BeanUtils.populate(t,map);  
                return t;  
            }catch(Exception e){  
                e.printStackTrace();  
            }  
        }  
        return null;  
    }  
}

package com.jsp.jdbc;  
  
import com.alibaba.druid.pool.DruidDataSource;  
import com.jsp.jdbc.handler.*;  
  
import java.sql.*;  
import java.util.Properties;  
  
public class JdbcUtil {  
    private static final DruidDataSource ds = new DruidDataSource();  
    public static void initDruidDataSource(Properties prop){//初始化数据源  
        ds.configFromPropety(prop);  
    }  
    public static void destroyDruidDataSource(){  
        ds.close();  
    }  
    public static <T> T Query(String sql , ResultHandler<T> handler, Object... params){  
        Connection conn = null;  
        try {  
            conn = ds.getConnection();  
            PreparedStatement ps = conn.prepareStatement(sql);  
            if(params != null){  
                for(int i = 0; i < params.length; i++){  
                    ps.setObject(i + 1 , params[i]);  
                }  
            }  
            ResultSet rs = ps.executeQuery();  
            T t = handler.handle(rs);//处理结果以前是进行遍历，反射，现在是写了一个方法类来解决处理  
           rs.close();  
           ps.close();  
           conn.close();  
           return t;  
        } catch (Exception e) {  
            throw new RuntimeException(e);  
        }  
    }  
    public static int Query(String sql , Object... params) {  
        Connection conn = null;  
        try {  
            conn = ds.getConnection();  
            PreparedStatement ps = conn.prepareStatement(sql);  
            if (params != null) {  
                for (int i = 0; i < params.length; i++) {  
                    ps.setObject(i + 1, params[i]);  
                }  
            }  
            int affected_Rows = ps.executeUpdate();  
            return affected_Rows;  
        } catch (Exception e) {  
            e.printStackTrace();  
            try {  
                conn.close();  
            } catch (Exception e1) {  
            }  
        }  
        return -1;  
    }  
}

package com.jsp.listener;  
  
import com.jsp.jdbc.JdbcUtil;  
  
import javax.servlet.ServletContext;  
import javax.servlet.ServletContextEvent;

    @Override

        System.out.println("监听器关闭释放资源");

    public void contextInitialized(ServletContextEvent sce) {

package com.jsp.Servlet;  
  
import com.alibaba.fastjson.JSONObject;  
import com.jsp.pojo.Student;  
import com.jsp.service.StudentService;  
import com.jsp.service.impl.StudentServiceImpl;  
  
import javax.servlet.ServletException;  
import javax.servlet.annotation.WebServlet;  
import javax.servlet.http.HttpServlet;  
import javax.servlet.http.HttpServletRequest;  
import javax.servlet.http.HttpServletResponse;  
import java.io.IOException;  
import java.io.PrintWriter;  
import java.util.ArrayList;  
import java.util.List;  
  
@WebServlet("/search")  
public class SearchServlet extends HttpServlet {  
  
    StudentService studentService = new StudentServiceImpl();  
  
    public void doGet(HttpServletRequest request, HttpServletResponse response)throws ServletException, IOException {  
        List<Student> students = new ArrayList<Student>();  
        students = studentService.searchStudent();  
        Object o = JSONObject .toJSON(students);  
        response.setContentType("text/html;charset=utf-8");  
        PrintWriter out = response.getWriter();  
        out.print(o);  
        out.flush();  
        out.close();  
    }  
}
