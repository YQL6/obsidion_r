package com.jsp.com.jsp.tools;  
  
import java.io.*;  
import java.util.*;  
  
import com.alibaba.excel.EasyExcel;  
import com.alibaba.excel.ExcelWriter;  
import com.alibaba.excel.context.AnalysisContext;  
import com.alibaba.excel.read.listener.ReadListener;  
import com.alibaba.excel.write.metadata.WriteSheet;  
import com.jsp.pojo.student;  
  
public class excel {  
    private static final int MAX_COUNT_PER_SHEET = 50;  
    public static void main(String[] args)throws Exception {  
        String dir = "D:\\JSP项目\\jsp07\\Excel示例.xlsx";  
        List<student> list = new ArrayList<>();  
        InputStream is = new FileInputStream(dir);  
        OutputStream os = new FileOutputStream(dir);  
        for(int i = 0 ; i <= 1000 ; i++){  
            student s = new student();  
            s.setAge(i);  
            s.setName("姓名" + i);  
            s.setSex(i % 2 == 0 ? "男" : "女");  
            list.add(s);  
        }  
        //ReadExcel(is , student.class);  
        WriteExcel(os , "sheet" , list , student.class);  
  
    }  
    public static<T> void WriteExcel(OutputStream os ,String sheetName , List<T> list , Class<T> clzz){  
//        EasyExcel.write(os,clzz).sheet(sheetName)  
//                .doWrite(list);  
        int size = list.size();  
        ExcelWriter excelWriter = EasyExcel.write(os, clzz).build();//创建Excel流  
        int sheetCount = size / MAX_COUNT_PER_SHEET;  
        if(size % MAX_COUNT_PER_SHEET > 0){  
            sheetCount++;  
        }  
        for (int i = 0; i < sheetCount; i++) {  
            int start = i * MAX_COUNT_PER_SHEET;  
            int end = start + MAX_COUNT_PER_SHEET;  
            end = end > size-1 ? size - 1 : end;  
            List<T> list_excel_sheet = list.subList(start, end);  
            WriteSheet writeSheet = new WriteSheet();  
            writeSheet.setSheetNo(i);  
            writeSheet.setSheetName(sheetName + i);  
            excelWriter.write(list_excel_sheet, writeSheet);  
        }  
        excelWriter.finish();  
    }  
    public static<T> List<T> ReadExcel (InputStream is, Class<T> clzz) {  
        List<T> list_excel = new ArrayList<>();  
        ReadListener<T> listener = new ReadListener<T>(){  
            @Override  
            public void invoke(T data, AnalysisContext analysisContext) {//读取后每一条的监听，操作行为  
//                System.out.println("读到一条数据：" + data);  
                //省略不读，只要集合  
                list_excel.add(data);  
            }  
  
            @Override  
            public void doAfterAllAnalysed(AnalysisContext analysisContext) {//全部读完的行为  
//                System.out.println(list_excel.toArray().length);  
            }  
        };  
//        EasyExcel.read(is , clzz , listener).sheet().doRead();  
        EasyExcel.read(is,clzz,listener).doReadAll();//不指定sheet了，都读  
  
        return list_excel;  
    }  
  
}
