# 文件传输

使用的工具：
1. FileItemFactory:
	1. 实现类——DiskFileItemFactory:
		1. setSizeThreshold:设置阈值，超过放临时磁盘位置，否则放在内存中缓存
		2. setRepsitory：设置临时磁盘位置
		3. 放入ServletFileUpload类做创建FileItem的工具
	2. 作用：创建FileItem
2. File Item：
	1. 表单所有内容都是一个File item，提交的表单有DiskFileItemFactory创建对象，text，password也是一个FileItem
	2. 方法：
		1. item.getname等
3. ServletFileUpload：
	1. 构造器：传入FileItemFactory（可以设置FileItemFactory）或不传入自动创建一个FileItemFactory
	2. setHeaderEncoding：字符编码，如何读信息
	3. setFileSizeMax：设置各文件的大小，过大parse方法会抛出异常，解决上传文件  
	4. setSizeMax：设置form表单请求的总大小
	5. parseRequest(new ServletRequestContext(request));  ：<mark style="background: #FF5582A6;">一定要是new ServletRequestContext(request)传入</mark>，解析表单内容调用FileItemFactory生成FileItem
4. ServletFileUpload.isMultipartContent(request)：判断是否表单有文件上传
5. IOUtils：
	1. 作用：tomcat自带的工具
	2. copy：传入输入输出流自动复制


示例：
uploadFileServlet:
```java
package com.jsp.servlet;  
  
import org.apache.tomcat.util.http.fileupload.*;  
import org.apache.tomcat.util.http.fileupload.disk.DiskFileItemFactory;  
import org.apache.tomcat.util.http.fileupload.servlet.ServletFileUpload;  
import org.apache.tomcat.util.http.fileupload.servlet.ServletRequestContext;  
  
import javax.servlet.ServletException;  
import javax.servlet.annotation.WebServlet;  
import javax.servlet.http.HttpServlet;  
import javax.servlet.http.HttpServletRequest;  
import javax.servlet.http.HttpServletResponse;  
import java.io.*;  
import java.util.List;  
import org.apache.commons.io.IOUtils;  
  
@WebServlet("/upload")  
public class uploadFileServlet extends HttpServlet {  
    private static final String SAVE_DIR = "D:\\JSP项目\\jsp07\\out\\upload";  
    public void doPost(HttpServletRequest request, HttpServletResponse response)throws ServletException, IOException {  
        if (!ServletFileUpload.isMultipartContent(request)) {  
            throw new RuntimeException("未发现Multipart/form-data");  
        } else {  
            DiskFileItemFactory factory = new DiskFileItemFactory();  
            factory.setSizeThreshold(4 * 1024 * 1024); //基础单位为B，设置阈值，超过放在自己设置的磁盘的临时存放位置，未超过就放在内存中  
            factory.setRepository(new File("java.io.tmpdir"));//设置超过阈值文件的存储位置  
            factory.setDefaultCharset("UTF-8");//设置字符集编码  
            ServletFileUpload upload = new ServletFileUpload(factory);//创建上传文件的对象，获得与解析上传表单的内容  
            upload.setHeaderEncoding("UTF-8");  
            upload.setFileSizeMax(50 * 1024 * 1024);//设置各文件的大小，过大parse方法会抛出异常，解决上传文件  
            upload.setSizeMax(500 * 1024 * 1024);//设置form表单请求的总大小  
            try {  
                List<FileItem> items = upload.parseRequest(new ServletRequestContext(request));  
                for(FileItem item : items) {  
                    if(item.isFormField()){//判断是否为普通表单字段  
                        System.out.println(item.getName() + "=>" + item.getString());//普通的表单字段  
                    }else{//上传的表单字段  
                        File dir = new File(SAVE_DIR);  
                        if(!dir.exists()){  
                            dir.mkdirs();  
                        }  
                        File save_file = new File(SAVE_DIR ,item.getName());  
                        InputStream is = item.getInputStream();  
                        OutputStream os = new FileOutputStream(save_file);  
                        IOUtils.copy(is,os);  
                        IOUtils.closeQuietly(is);  
                        IOUtils.closeQuietly(os);  
                    }  
                }  
                response.setCharacterEncoding("UTF-8");  
                response.setContentType("text/html;characterUTF-8");  
                response.getWriter().print("上传成功");  
            }catch(FileUploadException e){  
                e.printStackTrace();  
                response.setCharacterEncoding("UTF-8");  
                response.getWriter().print("上传失败");  
            }  
        }  
    }  
}
```
