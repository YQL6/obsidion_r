# ajax封装操作

```java
function ajax(option){  
    option.errorcallback = undefined;  
    option.successcallback = undefined;  
    ajax1(option.method,option.url,option.data,option.contentType,option.successcallback,option.errorcallback);  
}  
function ajax1(method,url,data,contentType,successcallback,errorcallback){  
    let username = document.getElementById("username");  
    username.onblur = function(){  
        let val = username.value;  
        if(val !== ""){  
            let xmlHttprequest;  
            if(window.ActiveXObject){  
                xmlHttprequest = new ActiveXObject("Microsoft.XMLHTTP");  
            }else{  
                xmlHttprequest = new XMLHttpRequest();  
            }  
            if(method.toLowerCase() === 'get'){  
                let param="?";  
                let keys = Object.keys(data);//获得对象的属性名集合  
                keys.forEach(key =>{  
                    param += key + "=" + data[key] + "&";//data[key]根据属性名来获得值  
                })  
                param = param.substring(0,param.length - 1);  
                url +=param;  
            }  
            xmlHttprequest.open(method , url, true);  
            xmlHttprequest.open(method, url, true);  
            xmlHttprequest.onreadystatechange = function(){  
                let result = xmlHttprequest.responseText;  
                if(xmlHttprequest.readyState === 4){  
                    if(xmlHttprequest.status >= 200 && xmlHttprequest.status < 300){  
                        if(typeof successcallback === 'function'){  
                            successcallback(result);  
                        }  
                    }else{  
                        if(typeof errorcallback === 'function'){  
                            errorcallback(result);  
                        }  
                    }  
                }  
            }  
            if(contentType){  
                xmlHttprequest.setRequestHeader('content-Type',contentType);  
            }  
            let dataInfo = null;  
            if(method.toLowerCase === 'get'){  
                dataInfo = data;//不是以post，get类型发送  
                if((contentType) && contentType.indexof('application/x-www-form-urlencoded')){  
                    let param = "";  
                    let keys = object.keys(data);  
                    keys.forEach(key =>{  
                        param += key + "&";  
                    })  
                    param.substring(0,param.length - 1);  
                    dataInfo = param;  
                }  
            }  
            xmlHttprequest.send(dataInfo);  
        }  
  
    }  
}
```

注：
1. 可以将ajax1的参数统一为option
2. 调用时引入，传入对象（重写方法和属性）即可
