# 二.动态语句

## 2.1whereif

----

- 映射配置：
```xml
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE mapper
        PUBLIC "-//mybatis.org//DTD Mapper 3.0//EN"
        "https://mybatis.org/dtd/mybatis-3-mapper.dtd">
<mapper namespace="com.atguigu.MyBaits.mapper.EmployeeMapper">

    <select id="query" resultType="employee">
        <!--可能传入属性，可能不传入属性
            传入：就判断这个属性
            不传入：不判断
            使用where
            if判断传入的参数，最终是否添加语句
                test属性会进行比较判断，正常些比较语句即可
                    true:进行拼接sql语句
                    false把拼接标签内部的语句
                    语法：key 比较符号 值 and|or key 比较符号 值
                    大于小于符号不推荐使用，推荐使用html中的比较符号 &gt; / &lt;
            -->
        select * from t_emp
        <!--使用where关键字来处理：
            1.自动添加where关键字，where内部有一个if满足,自动添加where关键字,不满足则去掉where
            2.自动去掉多余的and和or关键字
        where

         若是两个同时不满足，则where无判断条件 ——》报错 -->
       <where>
           <if test="name != null">
               emp_name = #{name}
           </if>
           <!--拼接时第一个不满住时，第二个有一个and可能出现错误-->
           <if test="salary != null and salary &gt; 100">
               and emp_salary = #{salary};
           </if>
       </where>

    </select>
</mapper>
```

## 2.2set标签

---

- 只要用于更新时的标签

- 映射配置：
```xml
 <!--更新数据-->
    <update id="update">
        update t_emp
        <!--set标签自动去逗号-->
        <set>
            <if test="empname != null">
                emp_name = #{empname}
            </if>
            <if test="empsalary != null">
                ,emp_salary = #{empsalary}
            </if>
        </set>
        where    emp_id = #{empid}

    </update>
```

## 2.4trim

---

- 自定义指定前缀或去掉后缀

- 映射文件配置
```xml
    <!--
        trim:
            prefix：若是有一个if满足则增加前缀
            suffix:若是有一个if满足则增加一个整体的后缀
            prefixOverrides：若是有一个满足则对满足的语句进行前缀去除
            suffixOverrides:若是有一个满足则对满足的语句进行后缀去除
            -->
    <update id="updateTrim">
        update t_emp
        <trim prefix="set" prefixOverrides=",">
            <if test="empname != null">
                emp_name = #{empname}
            </if>
            <if test="empsalary != null">
                ,emp_salary = #{empsalary}
            </if>
        </trim>
        where    emp_id = #{empid}

    </update>
    <select id="queryTrim" resultType="employee">
        select * from t_emp
        <trim prefix="set" prefixOverrides="and|or">
            <if test="name != null">
                emp_name = #{name}
            </if>
            <!--拼接时第一个不满住时，第二个有一个and可能出现错误-->
            <if test="salary != null and salary &gt; 100">
                and emp_salary = #{salary};
            </if>
        </trim>
        
    </select>
```

## 2.5choose

---

- 多个条件中选一个

- 映射配置
```xml
<select id="queryChoose" resultType="employee">
        select * from t_emp
        <!--choose不会自动添加where关键字
	        当有一个执行条件满足时，不会向下执行，仅一个条件可以进入
        -->
        where
        <choose>
            <when test="name != null">
                emp_name = #{name}
            </when>
            <!--拼接时第一个不满住时，第二个有一个and可能出现错误-->
            <when test="salary != null and salary &gt; 100">
                and emp_salary = #{salary};
            </when>
            <otherwise>
                1=1
            </otherwise>
        </choose>
            

    </select>
```

## 2.6批量操作

---

- 动态拼接为一个`sql语句`进行操作
- 将语句放到缓冲区，等待一起发送给数据库，<mark style="background: #FF5582A6;">但是注意自动提交要关闭</mark>，若是不关闭，让`sql`语句等待的语句到数据库中还是一条一个提交，没有事务控制了而且依旧是多个事务耗时大，效率低

- 关闭事务提交的方法：
	- 在创建`sqlfaction`到`sqlSession`时传入`flase`参数，关闭事务提交
	- `mybatis`默认关闭自动提交

- 数据库的配置：
```xml
<dataSource type="POOLED">
                <!--环境配置-->
                <property name="driver" value="com.mysql.cj.jdbc.Driver"/>
                <property name="url" value="jdbc:mysql:///mybatis-example?allowMultiQueries=true"/>
                <property name="username" value="root"/>
                <property name="password" value="root"/>
            </dataSource>
```

- 映射文件的配置：
```xml
</select>
    <!--
        批量操作的数据如何取 —— 》 使用@Param标识集合形参的名字
        若是直接使用形参的名字取 ——》 取不到
        也可以使用默认的arg1...依次取参数，取到的是整个的参数,这个参数可能就是集合的入口
        -->
    <!--
    批量操作的结构：找到可以重复使用的结构，进行拼接
        select * from t_emp
        where
        emp_id in (1,2,3);
   为了得到这样的结构 ——》 拼接
    -->
    <select id="queryBatch" resultType="employee">
        select * from t_emp
        where
        emp_id in
        <!--
        <foreach collection="" close="" separator="" item="" open=""></foreach>
            collection:指定用的是哪个形参做为集合 使用@Param()指定
            open:遍历过程前想要添加的字符串
            close:遍历后想添加什么字符串
            separator:遍历过程中添加的字符串，若是最后一次添加，不适用
            item:获得每个遍历项 ——》 相当于从集合中获得的值的key
        -->
        <foreach collection="ids" open="(" close=")" separator="," item="id">
            <!--每次遍历添加的数值内容-->
            #{id}
        </foreach> ;
    </select>
    <delete id="deleteBatch">
        delete from t_emp where id id
            <foreach collection="ids" open="(" close=")" separator="," item="id">
                #{id}
            </foreach>
    </delete>
    <insert id="insertBatch">
        insert into t_emp (emp_name,emp_salary)
        values
        <!--
            values (xx,xxx),(xxx,xxx)
            没有前缀，直接就是对括号进行遍历，逗号隔开
            一个括号就是一个遍历
            -->
        <foreach collection="list" separator="," item="employee">
            <!--一次性遍历的数值就是一个括号-->
            (#{employee.empname},#{employee.empsalary})
        </foreach>
    </insert>
    <update id="updateBatch">
        <!--每一句的局部都无法遍历
            解决：使用整体的遍历
        -->
<!--        update t_emp set emp_name = #{empname} , emp_salary = #{empsalary}; -->
        <foreach collection="list" item="employee">
            <!--一个标签设计执行多个语句，要设置允许执行多语句，否则会报错
            和jdbc一致，将参数设置好插入url配置中
            -->
            update t_emp set emp_name = #{employee.empname} , emp_salary = #{employee.empsalary};
        </foreach>
    </update>
```

- 接口内容:
```java
package com.atguigu.MyBaits.mapper;

import com.atguigu.MyBaits.pojo.Employee;
import org.apache.ibatis.annotations.Param;

import java.util.List;

public interface EmployeeMapper {
    //根据员工的姓名和工资查询员工的信息
    List<Employee> query(@Param("name") String name,@Param("salary") Double salary);
    List<Employee> queryTrim(@Param("name") String name,@Param("salary") Double salary);
    List<Employee> queryChoose(@Param("name") String name,@Param("salary") Double salary);
    //根据员工id更新员工设计，要求传入的name和salary不为null时才更新
    int update(Employee employee);
    int updateTrim(Employee employee);

    //批量查询
    List<Employee> queryBatch(@Param("ids") List<Integer> ids);
    //批量删除
    int deleteBatch(@Param("ids") List<Integer> id);
    //批量插入
    int insertBatch(@Param("list") List<Employee> employee);
    //批量修改
    int updateBatch(@Param("list") List<Employee> employee);
}

```
